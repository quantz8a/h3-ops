  • cir — compile / validate / revise / emit ContextDoc
  • ops — doctor / gate / GPU lock / run / batch

git clone https://github.com/quantz8a/h3-ops.git && cd h3-ops
python3 -m venv .venv && source .venv/bin/activate
python -m pip install -e .
export H3_OPS_H3C_SRC=/path/to/h3.c
export H3_OPS_MODEL_DIR=/path/to/MiniMax-H3
h3ctl doctor
h3ctl run --preset smoke --prompt-file examples/smoke.prompt.txt -o out/smoke.mp4
